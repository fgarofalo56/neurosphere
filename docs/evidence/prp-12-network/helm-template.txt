# helm template neurosphere infra/helm -f infra/helm/ci/values-ci.yaml (NetworkPolicy documents only)
---
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: neurosphere-default-deny
  labels:
    app.kubernetes.io/name: neurosphere
    app.kubernetes.io/instance: neurosphere
    app.kubernetes.io/managed-by: Helm
    helm.sh/chart: neurosphere-0.1.0
spec:
  podSelector:
    matchLabels:
      app.kubernetes.io/instance: neurosphere
  policyTypes:
    - Ingress
    - Egress

---
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: neurosphere-allow-dns
  labels:
    app.kubernetes.io/name: neurosphere
    app.kubernetes.io/instance: neurosphere
    app.kubernetes.io/managed-by: Helm
    helm.sh/chart: neurosphere-0.1.0
spec:
  podSelector:
    matchLabels:
      app.kubernetes.io/instance: neurosphere
  policyTypes:
    - Egress
  egress:
    - to:
        - namespaceSelector:
            matchLabels:
              kubernetes.io/metadata.name: kube-system
          podSelector:
            matchLabels:
              k8s-app: kube-dns
      ports:
        - protocol: UDP
          port: 53
        - protocol: TCP
          port: 53
