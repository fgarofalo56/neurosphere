git tag -a v0.1.0-rc.2 -m "NeuroSphere v0.1.0-rc.2" 66f18d9
git push origin v0.1.0-rc.2                      # S8, operator-approved 2026-10-26

for svc in api workers frontend; do

    ghcr.io/fgarofalo56/neurosphere/$svc@sha256:<digest>

    --certificate-identity "$IDENTITY" --certificate-oidc-issuer "$ISSUER" \
    ghcr.io/fgarofalo56/neurosphere/$svc@sha256:<digest>
