Order: merge the PRP-32 PR first, then tag `main`. Each step after the merge needs approval.

```bash
# 1. Tag the merged main commit (local, no push yet)
git checkout main
git pull --ff-only
git tag -a v0.1.0-rc.2 -m "v0.1.0-rc.2"

# 2. Push the tag (gate S8, explicit operator approval required)
git push origin v0.1.0-rc.2

# 3. Watch the release workflow
gh run list --workflow release.yml --limit 1
gh run watch <run-id> --exit-status

# 4. Look up digests (never verify the mutable tag)
for svc in api workers frontend; do
  docker buildx imagetools inspect ghcr.io/fgarofalo56/neurosphere/$svc:0.1.0-rc.2
done

# 5. Verify signatures and SBOM attestations per docs/release/verify.md
IDENTITY="https://github.com/fgarofalo56/neurosphere/.github/workflows/release.yml@refs/tags/v0.1.0-rc.2"
ISSUER="https://token.actions.githubusercontent.com"

for svc in api workers frontend; do
  cosign verify \
    --certificate-identity "$IDENTITY" \
    --certificate-oidc-issuer "$ISSUER" \
    ghcr.io/fgarofalo56/neurosphere/$svc@sha256:<digest>

  cosign verify-attestation --type cyclonedx \
    --certificate-identity "$IDENTITY" \
    --certificate-oidc-issuer "$ISSUER" \
    ghcr.io/fgarofalo56/neurosphere/$svc@sha256:<digest>
done
```

Record only identity, issuer and exit codes in sections 3 and 4.
